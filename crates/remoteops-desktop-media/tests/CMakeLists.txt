cmake_minimum_required(VERSION 3.25)
project(remoteops_clipboard_tests LANGUAGES CXX)
include(CTest)
function(clipboard_probe target worker)
    add_executable(${target} native_clipboard.cpp ${worker} ../native/clipboard_ole.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(${target} PRIVATE user32 shell32 ole32 uuid windowscodecs gdi32)
endfunction()
clipboard_probe(clipboard_station ../native/clipboard.cpp)
clipboard_probe(clipboard_fixture clipboard_fixture_worker.cpp)
target_compile_definitions(clipboard_fixture PRIVATE RO_CLIPBOARD_FIXTURE)
add_test(NAME clipboard_fixture COMMAND clipboard_fixture)
