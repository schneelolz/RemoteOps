cmake_minimum_required(VERSION 3.20)
project(remoteops_view_decoder LANGUAGES CXX)
if(WIN32)
  add_library(remoteops_view_decoder SHARED ../../../crates/remoteops-desktop-media/native/codec.cpp decoder.def)
  target_compile_options(remoteops_view_decoder PRIVATE /utf-8 /EHsc)
  target_link_libraries(remoteops_view_decoder PRIVATE ole32 oleaut32 mf mfplat mfuuid wmcodecdspuuid)
elseif(APPLE)
  enable_language(OBJCXX)
  add_library(remoteops_view_decoder SHARED decoder.mm)
  target_compile_options(remoteops_view_decoder PRIVATE -fobjc-arc)
  target_link_libraries(remoteops_view_decoder PRIVATE "-framework Foundation" "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo")
else()
  message(FATAL_ERROR "本控制端支持 Windows 和 macOS")
endif()
target_compile_features(remoteops_view_decoder PRIVATE cxx_std_17)
install(TARGETS remoteops_view_decoder RUNTIME DESTINATION . LIBRARY DESTINATION .)
